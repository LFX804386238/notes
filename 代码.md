    // ==================== 5. 渲染区块 ④：今日时间轴 (比例间距 + 实时「现在」蓝点) ====================
    function formatMinutesToHHMM(totalMin) {
      const h = Math.floor(totalMin / 60) % 24;
      const m = totalMin % 60;
      return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
    }

    function formatIntervalLabel(diffMin) {
      if (diffMin <= 0) return '＋0 分';
      const h = Math.floor(diffMin / 60);
      const m = diffMin % 60;
      if (h > 0 && m > 0) return `＋${h} 小时 ${m} 分`;
      if (h > 0) return `＋${h} 小时`;
      return `＋${m} 分`;
    }

    function getCurrentTimelineMinutes() {
      if (!state.useRealSystemClock) {
        return state.demoTimeMinutes; // 08:45 (525 分钟)
      }
      const now = new Date();
      return now.getHours() * 60 + now.getMinutes();
    }

    function renderTimelines() {
      const nowMin = getCurrentTimelineMinutes();
      const nowStr = formatMinutesToHHMM(nowMin);

      // 更新顶部时钟与说明
      document.getElementById('headerClock').textContent = nowStr;
      document.getElementById('timelineModeHint').textContent = state.useRealSystemClock
        ? `实时 · ${nowStr}`
        : `基准 · 08:45`;

      // 合并固定事件与「现在」蓝点，并按分钟数升序排列
      const nowNode = {
        timeMin: nowMin,
        timeStr: nowStr,
        title: '现在',
        badge: '你在这里',
        type: 'now',
        desc: '实时时间定位点'
      };

      const allNodes = [...baseTimelineEvents, nowNode].sort((a, b) => a.timeMin - b.timeMin);

      // ---------- A. 渲染手机端竖排真实比例时间轴 ----------
      const mobileContainer = document.getElementById('mobileVerticalTimeline');
      let mobileHtml = '';

      // 真实时间比例尺：每 1 分钟对应 0.45px（仅保留防文字重叠的 24px 最小触控安全线，不设上限封顶）
      const PX_PER_MINUTE = 0.45;
      const MIN_GAP_PX = 24;

      allNodes.forEach((node, index) => {
        // 节点样式映射
        let nodeCircle = '';
        let badgeStyle = '';
        let rowHighlight = '';

        if (node.type === 'success') {
          nodeCircle = `<span class="w-4 h-4 rounded-full bg-emerald-500 ring-3 ring-emerald-500/15 flex items-center justify-center text-[9px] text-white font-bold">✓</span>`;
          badgeStyle = 'bg-emerald-100/90 text-emerald-700 dark:bg-emerald-950/80 dark:text-emerald-300';
        } else if (node.type === 'error') {
          nodeCircle = `<span class="w-4 h-4 rounded-full bg-red-500 ring-3 ring-red-500/20 dot-danger-pulse flex items-center justify-center text-[9px] text-white font-bold">✗</span>`;
          badgeStyle = 'bg-red-100 text-red-700 dark:bg-red-950/80 dark:text-red-300 font-semibold';
          rowHighlight = 'bg-red-50/70 dark:bg-red-950/25 px-2 py-1 rounded-lg border border-red-200/60 dark:border-red-900/50';
        } else if (node.type === 'now') {
          nodeCircle = `<span class="w-4 h-4 rounded-full bg-blue-600 dark:bg-blue-500 ring-4 ring-blue-500/25 dot-now-pulse flex items-center justify-center"><span class="w-1.5 h-1.5 rounded-full bg-white"></span></span>`;
          badgeStyle = 'bg-blue-600 text-white dark:bg-blue-500 font-semibold shadow-2xs';
          rowHighlight = 'bg-blue-50/85 dark:bg-blue-950/35 border border-blue-200/80 dark:border-blue-800/60 px-2 py-1.5 rounded-lg';
        } else {
          // pending 待跑
          nodeCircle = `<span class="w-3.5 h-3.5 rounded-full bg-white dark:bg-obsidian-card border-2 border-slate-300 dark:border-slate-600"></span>`;
          badgeStyle = 'bg-slate-100 text-slate-500 dark:bg-slate-800 dark:text-slate-400';
        }

        const defaultRowPad = rowHighlight ? rowHighlight : 'px-2 py-0.5';

        // 渲染当前节点行
        mobileHtml += `
          <div class="flex items-center justify-between gap-2 ${defaultRowPad}">
            <div class="flex items-center gap-2.5 min-w-0">
              <span class="w-10 font-mono text-[12px] font-bold tabular-nums shrink-0 ${
                node.type === 'now' ? 'text-blue-600 dark:text-blue-400' :
                node.type === 'error' ? 'text-red-600 dark:text-red-400' :
                'text-slate-700 dark:text-slate-300'
              }">${node.timeStr}</span>

              <div class="w-5 flex justify-center shrink-0">
                ${nodeCircle}
              </div>

              <span class="text-[13px] font-bold truncate ${
                node.type === 'now' ? 'text-blue-700 dark:text-blue-300' :
                node.type === 'error' ? 'text-red-700 dark:text-red-300' :
                node.type === 'pending' ? 'text-slate-500 dark:text-slate-400' :
                'text-slate-900 dark:text-slate-100'
              }">${node.title}</span>

              ${node.badge}

          </div>
        `;

        // 渲染节点之间的真实比例间隔线（完全随分钟差线性拉伸，不固定高度）
        if (index < allNodes.length - 1) {
          const nextNode = allNodes[index + 1];
          const diffMin = Math.max(0, nextNode.timeMin - node.timeMin);
          const connectorHeight = Math.max(MIN_GAP_PX, Math.round(MIN_GAP_PX + diffMin * PX_PER_MINUTE));
          const isFutureSegment = node.type === 'now' || node.type === 'pending';

          mobileHtml += `
            <div class="flex items-center gap-2.5 px-2 transition-all duration-300" style="height: ${connectorHeight}px;">
              <span class="w-10 shrink-0"></span>

                <div class="w-0.5 h-full ${
                  isFutureSegment
                    ? 'border-l-2 border-dashed border-slate-200 dark:border-slate-700'
                    : 'bg-slate-200 dark:bg-slate-700'
                }"></div>
              </div>
              <span class="text-[10px] font-mono tabular-nums px-1.5 py-0.5 rounded bg-slate-100/90 dark:bg-slate-800/80 text-slate-500 dark:text-slate-400 leading-none">
                ${formatIntervalLabel(diffMin)}
              </span>
            </div>
          `;
        }
      });

      mobileContainer.innerHTML = mobileHtml;

      // ---------- B. 渲染桌面端 00:00–24:00 横轴带刻度 ----------
      const desktopContainer = document.getElementById('desktopHorizontalTimeline');
      const ticks = [0, 3, 6, 9, 12, 15, 18, 21, 24];

      const ticksHtml = ticks.map(h => {
        const leftPct = (h / 24) * 100;
        return `
          <div class="absolute top-0 -translate-x-1/2 flex flex-col items-center" style="left: ${leftPct}%;">

          </div>
        `;
      }).join('');

      const nodesHtml = allNodes.map((node, idx) => {
        const leftPct = Math.min(98, Math.max(2, (node.timeMin / 1440) * 100));
        // 上下交替错开标签防止 08:36 / 08:45 / 08:50 密集重叠
        const isTop = idx % 2 === 0;
        let dotBg = 'bg-slate-300 dark:bg-slate-600';
        let labelColor = 'text-slate-600 dark:text-slate-300 bg-slate-100 dark:bg-slate-800';

        if (node.type === 'success') {
          dotBg = 'bg-emerald-500 ring-4 ring-emerald-500/20';
          labelColor = 'text-emerald-700 dark:text-emerald-300 bg-emerald-50 dark:bg-emerald-950/80 border border-emerald-200 dark:border-emerald-800';
        } else if (node.type === 'error') {
          dotBg = 'bg-red-500 ring-4 ring-red-500/25 dot-danger-pulse';
          labelColor = 'text-red-700 dark:text-red-300 bg-red-50 dark:bg-red-950/80 border border-red-200 dark:border-red-800 font-semibold';
        } else if (node.type === 'now') {
          dotBg = 'bg-blue-600 ring-4 ring-blue-500/30 dot-now-pulse';
          labelColor = 'text-white bg-blue-600 font-semibold shadow-sm';
        }

        return `
          <div class="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 group" style="left: ${leftPct}%;">
            <div class="w-3.5 h-3.5 rounded-full ${dotBg} cursor-pointer"></div>
            <div class="absolute left-1/2 -translate-x-1/2 whitespace-nowrap px-2 py-0.5 rounded-md text-[10px] font-mono ${labelColor} ${
              isTop ? 'bottom-5' : 'top-5'
            }">
              ${node.timeStr} ${node.title} · ${node.badge}
            </div>
          </div>
        `;
      }).join('');

      const nowPct = Math.min(100, Math.max(0, (nowMin / 1440) * 100));

      desktopContainer.innerHTML = `
        <div class="relative h-24 px-4 select-none">
          <!-- 主时间轴轨道 -->
          <div class="relative top-1/2 -translate-y-1/2 h-1.5 w-full rounded-full bg-slate-200 dark:bg-slate-800">
            <!-- 已流逝时间进度条 -->
            <div class="h-full rounded-full bg-blue-500/40 dark:bg-blue-500/30" style="width: ${nowPct}%;"></div>
            ${nodesHtml}
          </div>
          <!-- 00:00 - 24:00 整点刻度 -->
          <div class="relative top-14 w-full">
            ${ticksHtml}
          </div>
        </div>
      `;
    }
