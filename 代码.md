        { timeMin: 8 * 60 + 36, timeStr: '08:36', title: '榜单', statusText: '✗ 失败', type: 'error' },
        {
          timeMin: 8 * 60 + 50,
          timeStr: '08:50',
          title: '日报',
          statusText: (nowMin >= 8 * 60 + 50 && state.clockMode !== 'demo') ? '✓ 已跑' : '待跑',
          type: (nowMin >= 8 * 60 + 50 && state.clockMode !== 'demo') ? 'ok' : 'pending'
        },
        {
          timeMin: 10 * 60 + 30,
          timeStr: '10:30',
          title: '补发',
          statusText: (nowMin >= 10 * 60 + 30 && state.clockMode !== 'demo') ? '✓ 已跑' : '待跑',
          type: (nowMin >= 10 * 60 + 30 && state.clockMode !== 'demo') ? 'ok' : 'pending'
        },
        {
          timeMin: 13 * 60 + 44,
          timeStr: '13:44',
          title: '证书',
          statusText: (nowMin >= 13 * 60 + 44 && state.clockMode !== 'demo') ? '✓ 已跑' : '待跑',
          type: (nowMin >= 13 * 60 + 44 && state.clockMode !== 'demo') ? 'ok' : 'pending'
        }
      ];

      const nowNode = {
        timeMin: nowMin,
        timeStr: nowStr,
        title: '现在',
        statusText: '你在这里',
        type: 'now'
      };

      const timelineNodes = [...baseEvents, nowNode].sort((a, b) => {
        if (a.timeMin !== b.timeMin) return a.timeMin - b.timeMin;
        return a.type === 'now' ? 1 : -1;
      });

      // ---------- A. 渲染手机端竖排比例时间轴 ----------
      const mobileContainer = document.getElementById('mobileTimelineWrapper');
      mobileContainer.innerHTML = timelineNodes.map((node, idx) => {
        let dotHtml = '';
        let rowCardClass = '';
        let badgeHtml = '';

        if (node.type === 'ok') {
          dotHtml = `<span class="w-3 h-3 rounded-full bg-emerald-500 ring-4 ring-emerald-500/15 shrink-0 z-10"></span>`;
          rowCardClass = 'text-slate-700 dark:text-slate-200';
          badgeHtml = `<span class="text-xs font-semibold text-emerald-600 dark:text-emerald-400">${node.statusText}</span>`;
        } else if (node.type === 'error') {
          dotHtml = `<span class="relative w-3 h-3 rounded-full bg-red-500 ring-4 ring-red-500/20 shrink-0 z-10 pulse-red"></span>`;
          rowCardClass = 'bg-red-50/90 dark:bg-red-950/35 border border-red-200/80 dark:border-red-900/70 rounded-lg px-2.5 py-1.5 text-red-700 dark:text-red-300';
          badgeHtml = `<span class="text-xs font-bold text-red-600 dark:text-red-400">${node.statusText}</span>`;
        } else if (node.type === 'now') {
          dotHtml = `<span class="relative w-3.5 h-3.5 rounded-full bg-blue-600 ring-4 ring-blue-500/25 shrink-0 z-10 pulse-blue"></span>`;
          rowCardClass = 'bg-blue-600 text-white rounded-lg px-2.5 py-1.5 shadow-sm';
          badgeHtml = `<span class="text-[11px] font-semibold bg-white/20 px-1.5 py-0.5 rounded">${node.statusText}</span>`;
        } else {
          dotHtml = `<span class="w-3 h-3 rounded-full bg-slate-300 dark:bg-slate-600 ring-4 ring-slate-200/50 dark:ring-slate-800 shrink-0 z-10"></span>`;
          rowCardClass = 'text-slate-500 dark:text-slate-400';
          badgeHtml = `<span class="text-xs font-medium text-slate-400 dark:text-slate-500">${node.statusText}</span>`;
        }

        let connectorHtml = '';
        if (idx < timelineNodes.length - 1) {
          const nextNode = timelineNodes[idx + 1];
          const diffMin = Math.max(0, nextNode.timeMin - node.timeMin);
          const intervalLabel = formatDeltaChinese(diffMin);

          const scaledHeight = Math.min(78, Math.max(26, Math.round(22 + diffMin * 0.24)));
          const isPassedSegment = nextNode.timeMin <= nowMin;
          const lineClass = isPassedSegment
            ? 'border-l-2 border-solid border-slate-300 dark:border-slate-600'
            : 'border-l-2 border-dashed border-slate-200 dark:border-slate-700';

          connectorHtml = `
            <div class="flex items-center pl-[5px]" style="height: ${scaledHeight}px;">
              <div class="h-full ${lineClass} mr-4"></div>
              <span class="text-[10px] font-mono px-1.5 py-0.5 rounded bg-slate-100/90 dark:bg-slate-800/80 text-slate-400 dark:text-slate-500">
                ${intervalLabel}
              </span>
            </div>
          `;
        }

        return `
          <div>
            <div class="flex items-center gap-2.5">
              ${dotHtml}
              <div class="flex-1 flex items-center justify-between gap-2 min-w-0 ${rowCardClass}">
                <div class="flex items-center gap-2 min-w-0">
                  <span class="font-mono text-xs font-bold tabular-nums shrink-0">${node.timeStr}</span>
                  <span class="text-xs font-semibold truncate">${node.title}</span>
                </div>
                <div class="shrink-0">
                  ${badgeHtml}
                </div>
              </div>
            </div>
            ${connectorHtml}
          </div>
        `;
      }).join('');

      // ---------- B. 渲染电脑端 00:00–24:00 横轴带刻度 ----------
      const desktopContainer = document.getElementById('desktopTimelineWrapper');
      const majorHours = [0, 3, 6, 9, 12, 15, 18, 21, 24];

      const ticksHtml = majorHours.map(h => {
        const leftPct = (h / 24) * 100;
        return `
          <div class="absolute top-0 -translate-x-1/2 flex flex-col items-center" style="left: ${leftPct}%;">
            <div class="w-px h-2.5 bg-slate-300 dark:bg-slate-600"></div>
            <span class="mt-1 text-[11px] font-mono text-slate-400 dark:text-slate-500">${String(h).padStart(2, '0')}:00</span>
          </div>
        `;
      }).join('');

      const tierLayouts = [
        { pos: 'above-1', stemH: 'h-6', boxOffset: '-top-12' },
        { pos: 'below-1', stemH: 'h-6', boxOffset: 'top-7' },
        { pos: 'above-2', stemH: 'h-12', boxOffset: '-top-[74px]' },
        { pos: 'below-2', stemH: 'h-12', boxOffset: 'top-[52px]' },
        { pos: 'above-1', stemH: 'h-6', boxOffset: '-top-12' },
        { pos: 'below-1', stemH: 'h-6', boxOffset: 'top-7' },
        { pos: 'above-1', stemH: 'h-6', boxOffset: '-top-12' }
      ];

      const nodesHtml = timelineNodes.map((node, idx) => {
        const leftPct = Math.min(97, Math.max(3, (node.timeMin / 1440) * 100));
        const tier = tierLayouts[idx % tierLayouts.length];
        const isAbove = tier.pos.startsWith('above');

        let dotClass = 'bg-slate-300 dark:bg-slate-600';
        let cardClass = 'bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-600 dark:text-slate-300';

        if (node.type === 'ok') {
          dotClass = 'bg-emerald-500 ring-4 ring-emerald-500/20';
          cardClass = 'bg-emerald-50 dark:bg-emerald-950/90 border-emerald-200 dark:border-emerald-800 text-emerald-800 dark:text-emerald-200';
        } else if (node.type === 'error') {
          dotClass = 'bg-red-500 ring-4 ring-red-500/25 pulse-red';
          cardClass = 'bg-red-50 dark:bg-red-950/95 border-red-300 dark:border-red-700 text-red-700 dark:text-red-200 font-bold shadow-xs';
        } else if (node.type === 'now') {
          dotClass = 'bg-blue-600 ring-4 ring-blue-500/30 pulse-blue';
          cardClass = 'bg-blue-600 border-blue-500 text-white font-bold shadow-sm';
        }

        const stemPosition = isAbove ? 'bottom-1/2' : 'top-1/2';

        return `
          <div class="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 z-10 group" style="left: ${leftPct}%;">
            <div class="absolute left-1/2 -translate-x-1/2 ${stemPosition} w-px ${tier.stemH} bg-slate-300 dark:bg-slate-600"></div>
            <div class="relative w-3.5 h-3.5 rounded-full ${dotClass} transition-transform group-hover:scale-125 cursor-pointer"></div>
            <div class="absolute left-1/2 -translate-x-1/2 ${tier.boxOffset} whitespace-nowrap px-2 py-0.5 rounded-md border text-[11px] font-mono ${cardClass}">
              ${node.timeStr} ${node.title} ${node.type === 'now' ? '· 你在这里' : node.statusText}
            </div>
          </div>
        `;
      }).join('');

      const elapsedPct = Math.min(100, Math.max(0, (nowMin / 1440) * 100));

      desktopContainer.innerHTML = `
        <div class="relative h-48 px-4 select-none">
          <div class="relative top-1/2 -translate-y-1/2 h-1.5 w-full rounded-full bg-slate-200 dark:bg-slate-800">
            <div class="h-full rounded-full bg-teal-500/40 dark:bg-teal-500/30 transition-all duration-300" style="width: ${elapsedPct}%;"></div>
            ${nodesHtml}
          </div>
          <div class="relative top-[156px] w-full">
            ${ticksHtml}
          </div>
        </div>
      `;
    }

    // 7. 统一渲染与实时时钟刷新
    function renderAll() {
      syncClockUI();
      renderTaskList();
      renderCloudSync();
      renderTimelines();
    }

    window.addEventListener('DOMContentLoaded', () => {
      renderAll();
      setInterval(() => {
        if (state.clockMode === 'live') {
          renderAll();
        }
      }, 15000);
    });
  </script>
