    function renderGauges() {
      const grid = document.getElementById('gaugesGrid');
      const radius = 38;
      const circumference = 2 * Math.PI * radius;

      grid.innerHTML = STATE.gauges.map((g) => {
        const clamped = Math.max(0, Math.min(100, g.pct));
        const offset = circumference - (clamped / 100) * circumference;
        const colorCls = getGaugeClass(clamped);
        return `
          <div class="gauge-card">
            <svg class="gauge-svg" viewBox="0 0 96 96" aria-label="${g.label} ${clamped}%">
              <circle class="gauge-track" cx="48" cy="48" r="${radius}" />
              <circle
                class="gauge-fill ${colorCls}"
                cx="48"
                cy="48"
                r="${radius}"
                stroke-dasharray="${circumference.toFixed(1)}"

            <div class="gauge-label">${g.label}</div>
            <div class="gauge-pct">${clamped}%</div>

    }
